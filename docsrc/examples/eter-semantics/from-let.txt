// NOTE: `let` is the default (implicit) regime — equivalent to `fix`.
// A bare `let x = ...` binding is immutable and freely copyable in O(1).
// Transition rules are identical to `fix`: let -> let is always allowed,
// let -> mut and let -> proj require an explicit copy.

// ==========
// Let to Let
// ==========
fn f(let x: i32) {}
fn main() {
    let x: i32 = 1;
    let y: i32 = x; // O(1)
    f(x);           // O(1)
}

// ==========
// Let to Mut
// ==========
fn f(mut x: i32) {}
fn main() {
    let x: i32 = 1;
    let mut y: i32 = x; // ERROR: Changing the regime from let to mut is not allowed. Need to copy.
    f(x);               // ERROR: Changing the regime from let to mut is not allowed. Need to copy.
}

// ===========
// Let to Proj
// ===========
fn f(proj x: i32) {}
fn main() {
    let x: i32 = 1;
    let proj y: i32 = &x; // ERROR: Changing the regime from let to proj is not allowed. Need to copy.
    f(&x);                // ERROR: Changing the regime from let to proj is not allowed. Need to copy.
}

struct Point {
    let x: i32,
    let y: i32,
}

// ==========
// Let to Let
// ==========
fn f(let p: Point) {}
fn main() {
    let p = Point { x: 1, y: 2 };
    let q = p;  // O(1)
    f(p);       // O(1)
}

// ==========
// Let to Mut
// ==========
fn f(mut p: Point) {}
fn main() {
    let p = Point { x: 1, y: 2 };
    let mut q = p; // ERROR: Changing the regime from let to mut is not allowed. Need to copy.
    f(p);          // ERROR: Changing the regime from let to mut is not allowed. Need to copy.
}

// ===========
// Let to Proj
// ===========

    f(&p);           // ERROR: Changing the regime from let to proj is not allowed. Need to copy.
}

// ========================================================================
// ========================================================================
// Store Let in Struct from Let
// ========================================================================
// ========================================================================

// ==========
// SINGLE CASE
// ==========
struct Point {
    let x: i32,
    let y: i32,
}

fn f(let p: Point): Point {
    let x = p.x;
    let y = p.y;
    Point { x, y }
}

fn tmp1() {
    let p1 = Point { x: 1, y: 2 };
    let p2 = f(p1);
}

// OPTIMIZATION: to avoid copy and reference counting
fn tmp2() {
    let p2;
    {
        let p1 = Point { x: 1, y: 2 };
        p2 = f(p1);
    }
}

// ==========
// Let-Let-Let as return and formal parameters from Let
// =========
struct Point {
    let x: i32,
    let y: i32,
}

fn f(let p: Point, let u: Point): Point {
    // u is not used
    let x = p.x;
    let y = p.y;
    Point { x, y }
}

fn tmp1() {
    let p1 = Point { x: 1, y: 2 };
    let p2 = Point { x: 3, y: 4 };
    let p3 = f(p1, p2);
    // - p1 can be used again
    // - p2 can be used again
}

fn tmp2() {
    let p3;
    let p = Point { x: 1, y: 2 };
    {
        let u = Point { x: 3, y: 4 };
        p3 = f(p, u);
    }
}

// ==========
// Let-Let-Mut as return and formal parameters from Let

// Store Mut in Struct from Let
// ========================================================================
// ========================================================================

// NOTE: let binding + mut field -> let (fix) access.
// A `let` binding propagates immutability: mut fields cannot be accessed as mut
// from a `let` binding.

// ==========
// SINGLE CASE
// ==========
struct Point {
    mut x: i32,
    mut y: i32,
}

fn f(let p: Point): Point {
    let x = p.x; // let (fix) access — let propagates to mut fields
    let y = p.y;
    // let mut x = p.x; // ERROR: p is under the regime let, it propagates to the whole struct.
    Point { x, y }
}

fn tmp1() {
    let p1 = Point { x: 1, y: 2 };
    let p2 = f(p1);
}

fn tmp2() {
    let p2;
    {
        let p1 = Point { x: 1, y: 2 };
        p2 = f(p1);
    }
}

}

fn tmp1() {

    let p2 = f(p1);
}

fn tmp2() {
    let p2;
    {
        let p1 = Point { x: 1, y: 2 };
        p2 = f(p1);
    }
